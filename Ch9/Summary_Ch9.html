101למה צריך מערך

רוצים לקלוט מספרים ולהדפיס כמה מהם גדולים מהממוצע. בלי מערך צריך משתנה נפרד לכל מספר ותנאי נפרד לכל משתנה. עם מערך יש שם אחד לכל המספרים, ועוברים עליהם בלולאה. כדי לעבור ל-20 או ל-100 מספרים משנים רק את הקבוע SIZE.

const int SIZE = 5;
int[] numbers = new int[SIZE];
int sum = 0;
for (int i = 0; i < numbers.Length; i++)
{
    numbers[i] = int.Parse(Console.ReadLine());
    sum = sum + numbers[i];
}
double average = sum / (double)SIZE;

int countAboveAverage = 0;
for (int i = 0; i < numbers.Length; i++)
{
    if (numbers[i] > average)
    {
        countAboveAverage++;
    }
}

106העתקת מערך

108פעולה שמחזירה מערך

הטיפוס המוחזר של פעולה כזו הוא int[]. CreateAndReadArr יוצרת מערך בגודל שקיבלה, קולטת אליו ערכים ומחזירה אותו. ResizeArr יוצרת מערך חדש בגודל אחר ומעתיקה אליו את הערכים.

static int[] ResizeArr(int[] arr, int size)
{
    int[] tmp = new int[size];
    for (int i = 0; i < size; i++)
    {
        if (i < arr.Length)
        {
            tmp[i] = arr[i];
        }
    }
    return tmp;
}

numbers = ResizeArr(numbers, newSize);

110הערך הגדול ביותר והאינדקס שלו

מתחילים מהאיבר הראשון כמקסימום ובודקים את כל השאר. כשצריך לדעת איפה נמצא המקסימום, שומרים את האינדקס. מהאינדקס תמיד אפשר לקבל גם את הערך.

static int IndexOfMax(int[] arr)
{
    int maxIndex = 0;
    for (int i = 1; i < arr.Length; i++)
    {
        if (arr[i] > arr[maxIndex])
        {
            maxIndex = i;
        }
    }
    return maxIndex;
}

111היפוך סדר האיברים

הפעולה יוצרת מערך חדש באותו גודל. האיבר במקום i עובר למקום arr.Length - 1 - i. סכום האינדקסים של שני תאים סימטריים שווה תמיד ל-Length - 1.

static int[] Reverse(int[] arr)
{
    int[] res = new int[arr.Length];
    for (int i = 0; i < arr.Length; i++)
    {
        res[arr.Length - 1 - i] = arr[i];
    }
    return res;
}

112מערך פלינדרום

מערך פלינדרום נקרא אותו דבר משני הכיוונים. משתמשים בשני אינדקסים: left מתחיל בהתחלה ו-right בסוף, ושניהם מתקרבים למרכז. ברגע שנמצאו שני איברים שונים, התשובה היא לא.

static bool IsPalindrome(int[] arr)
{
    int left = 0;
    int right = arr.Length - 1;
    while (left < right)
    {
        if (arr[left] != arr[right])
        {
            return false;
        }
        left++;
        right--;
    }
    return true;
}
פלט
1 2 3 2 1 is palindrome: True
1 2 3 4 is palindrome: False

113סיבוב מערך שמאלה

הערך הראשון עובר לסוף, וכל שאר האיברים זזים מקום אחד שמאלה. שומרים קודם את האיבר הראשון במשתנה עזר, אחרת הוא יימחק.

static void RotateLeft(int[] arr)
{
    int first = arr[0];
    for (int i = 1; i < arr.Length; i++)
    {
        arr[i - 1] = arr[i];
    }
    arr[arr.Length - 1] = first;
}

שימו לב: הפעולה היא void ובכל זאת המערך המקורי משתנה, כי מערך שמועבר לפעולה הוא אותו מערך (דוגמה 107).

114התווים השונים במערך

מדפיסים כל תו פעם אחת וסופרים כמה תווים שונים יש. תו מודפס רק אם הוא לא הופיע באף מקום לפניו. לכל משימה יש פעולת עזר משלה.

static bool HadAppearBefore(char[] arr, int index)
{
    for (int i = 0; i < index; i++)
    {
        if (arr[i] == arr[index])
        {
            return true;
        }
    }
    return false;
}

static int CountAndShowUnique(char[] arr)
{
    int count = 0;
    for (int i = 0; i < arr.Length; i++)
    {
        if (HadAppearBefore(arr, i) == false)
        {
            Console.Write(arr[i] + " ");
            count++;
        }
    }
    Console.WriteLine();
    return count;
}

115מיזוג שני מערכים ממוינים

מערך ממוין הוא מערך שאיבריו מסודרים מהקטן לגדול. יוצרים מערך חדש בגודל של שניהם יחד. בכל שלב משווים את האיבר הנוכחי בכל מערך ומעתיקים את הקטן. כשאחד המערכים נגמר, מעתיקים את מה שנשאר בשני.

        res[m] = arr1[i];
        i++;
    }
    else
    {
        res[m] = arr2[j];

static double[] BuildMilkArray(int numCows, int days)
{
    double[] arr = new double[numCows];

    }
    return arr;
}

118חיפוש סדרתי

עוברים על האיברים אחד אחרי השני, עד שמוצאים את המספר או עד סוף המערך. החיפוש עובד גם במערך שאינו ממוין, אבל במערך גדול הוא לוקח זמן רב.

static bool Search(int[] arr, int number)
{
    for (int i = 0; i < arr.Length; i++)
    {
        if (arr[i] == number)
        {
            return true;
        }
    }
    return false;
}

119מיון בועות

משווים כל שני איברים שכנים, ואם הם לא בסדר הנכון מחליפים ביניהם. אחרי סיבוב שלם, הערך הגדול ביותר מגיע לסוף המערך, כמו בועה שעולה למעלה. בכל סיבוב הבא עוצרים מקום אחד קודם, כי הסוף כבר מסודר.

static void Swap(int[] arr, int i, int j)
{
    int temp = arr[i];
    arr[i] = arr[j];
    arr[j] = temp;
}

static void BubbleSort(int[] arr)
{
    for (int round = 0; round < arr.Length - 1; round++)
    {
        for (int i = 0; i < arr.Length - 1 - round; i++)
        {
            if (arr[i] > arr[i + 1])
            {
                Swap(arr, i, i + 1);
            }
        }
    }
}
פלט
Before: 5 1 4 2 8
After:  1 2 4 5 8

120מיון בחירה

בכל סיבוב מחפשים את הערך הקטן ביותר בחלק שעוד לא מוין, ומחליפים אותו עם האיבר הראשון של החלק הזה. כך החלק הממוין בתחילת המערך גדל באיבר אחד בכל סיבוב.

static void SelectionSort(int[] arr)
{
    for (int start = 0; start < arr.Length - 1; start++)
    {
        int minIndex = start;
        for (int i = start + 1; i < arr.Length; i++)
        {
            if (arr[i] < arr[minIndex])
            {
                minIndex = i;
            }
        }
        Swap(arr, start, minIndex);

static int BinarySearch(int[] arr, int number)
{
    int low = 0;
    int high = arr.Length - 1;
    while (low <= high)
    {
        int middle = (low + high) / 2;
        if (arr[middle] == number)
        {
            return middle;
        }
        if (number < arr[middle])
        {
            high = middle - 1;
        }
        else
        {
            low = middle + 1;
        }
    }
    return -1;
}
